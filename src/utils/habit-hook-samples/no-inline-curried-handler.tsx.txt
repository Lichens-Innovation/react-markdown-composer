import { useState } from "react";

export const Counter = (): JSX.Element => {
  const [count, setCount] = useState(0);
  const makeIncrementBy = (amount: number) => () => setCount((prev) => prev + amount);
  return <button onClick={makeIncrementBy(1)}>{count}</button>;
};
